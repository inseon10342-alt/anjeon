/** 휴대폰 앱(GitHub Pages) → 서버 호출: {c: 접속 코드, f: 함수 이름, a: [인자]} → {ok, v | e} */
function apiCall_(body) {
  const json = o => ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
  let req;
  try { req = JSON.parse(body); } catch (e) { return json({ ok: false, e: '잘못된 요청' }); }
  if (String(req.c || '') !== String(CONFIG.ACCESS_CODE)) return json({ ok: false, e: 'ACCESS:접속 코드가 맞지 않습니다' });
  if (req.f === 'ping') return json({ ok: true, v: 'pong' });
  const fn = API_FNS.indexOf(req.f) >= 0 ? API_IMPL()[req.f] : null;
  if (typeof fn !== 'function') return json({ ok: false, e: '알 수 없는 요청: ' + req.f });
  try { const v = fn.apply(null, Array.isArray(req.a) ? req.a : []); return json({ ok: true, v: v === undefined ? null : v }); }
  catch (err) { return json({ ok: false, e: String((err && err.message) || err) }); }
}

function doPost(e) {
  const out = s => ContentService.createTextOutput(s).setMimeType(ContentService.MimeType.TEXT);
  const body = (e && e.postData && e.postData.contents) || '';
  if (body.charAt(0) === '{') return apiCall_(body);               // 휴대폰 앱 (GitHub 주소)
  try {
    const req = parseTsvBody_(body);                               // 통합관리 엑셀 「현장앱 보내기」
    if (req.KEY !== CONFIG.API_KEY) return out('ERR\t키(API_KEY)가 맞지 않습니다');
    if (req.ACTION === 'ping') return out('OK\t연결 정상');
    if (req.ACTION !== 'push') return out('ERR\t알 수 없는 요청');
    return out('OK\t' + applyPush_(req.sections.SERVICES || [], req.sections.ITEMS || [], req.sections.CHECKLIST || [], '보내기 버튼'));
  } catch (err) {
    return out('ERR\t' + err.message);
  }
}

/** 통합관리 → 현장앱 반영 (보내기 버튼 · 자동 반영 공용) */
function applyPush_(svc, items, chk, how) {
  return withLock_(() => {
    writeTab_('svc', svc.filter(r => r[0]));
    writeTab_('insp', items.filter(r => r[0] && r[2]));
    if (chk.length) { writeTab_('chk', chk.filter(r => r[0] && !/^※/.test(r[0]))); formsFromTab_(); }
    setP_('LAST_PUSH', fmt_(new Date(), 'yyyy-MM-dd HH:mm') + (how ? ' (' + how + ')' : ''));
    bustBoot_();
    const act = svc.filter(r => r[0] && (!r[5] || r[5] === '진행중')).length;
    return `용역 ${svc.length}건 (진행중 ${act} · 끝난 용역 ${svc.length - act}) · 점검항목 ${items.length}건 · 체크리스트 ${chk.length}항목`;
  });
}

function parseTsvBody_(body) {
  const req = { sections: {} };
  let cur = null;
  String(body).replace(/\r/g, '').split('\n').forEach(line => {
    if (!line.trim()) return;
    if (line[0] === '#') { cur = line.slice(1).trim(); req.sections[cur] = []; return; }
    const p = line.split('\t').map(s => s.trim());
    if (cur) req.sections[cur].push(p); else req[p[0]] = p[1] || '';
  });
  return req;
}

  return forms;
}

/**
 * 휴대폰(기기)별 사용자 이름 등록 — 처음 한 번
 *  - 다른 기기에서 이미 쓰는 이름이면 taken 으로 알려 줌 (같은 사람이면 confirm=true 로 기기 추가)
 */
function registerUser(name, deviceId, confirmSame) {
  name = needUser_(name);
  deviceId = String(deviceId || '').slice(0, 64);
  if (!deviceId) throw new Error('기기 정보가 없습니다.');
  return withLock_(() => {
    const rows = readTab_('users');
    const now = fmt_(new Date(), 'yyyy-MM-dd HH:mm:ss');
    const mine = rows.find(r => r[0] === name && r[1] === deviceId);
    if (mine) return { ok: true, name };
    const others = rows.filter(r => r[0] === name);
    if (others.length && !confirmSame) return { taken: true, name, since: others[0][2], count: others.length };
    appendRow_('users', [name, deviceId, now]);
    log_(name, others.length ? '기기추가' : '사용자등록', { id: '', code: '', insp: '' }, deviceId.slice(0, 8), '');
    bustBoot_();
    return { ok: true, name };
  });
}

/** 목록 (사진 + 조사). 변경이 없으면 same:true 만 반환 → 가벼운 자동 갱신 */
function listItems(code, stamp) {
  const cur = stamp_(code);
  if (stamp && stamp === cur) return { same: true, stamp: cur };
  // 같은 변경분은 한 사람만 시트를 읽고, 나머지 휴대폰은 캐시에서 (사진이 많고 여러 명일 때 빠름)
  const ck = 'LI_' + code + '_' + cur;
  try {
    const n = +CACHE().get(ck);
    if (n > 0) {
      const keys = Array.from({ length: n }, (_, i) => ck + '_' + i), m = CACHE().getAll(keys);
      if (keys.every(k => m[k] != null)) {
        const r = JSON.parse(keys.map(k => m[k]).join(''));
        r.now = fmt_(new Date(), 'yyyy-MM-dd HH:mm:ss');
        return r;
      }
    }
  } catch (e) { /* 캐시 없음 → 시트에서 */ }
  const items = readTab_('items').map(rowToItem_).filter(o => o.code === code && !o.del);
  const surveys = readTab_('surveys').map(rowToSurvey_).filter(s => s.code === code && !s.del);
  const specs = readRecs_('specs').filter(x => x.code === code && !x.del);
  const measures = readRecs_('measures').filter(x => x.code === code && !x.del);
  const r = { stamp: cur, items, surveys, specs, measures };
  try {
    const js = JSON.stringify(r), parts = {};
    let n = 0;
    for (let i = 0; i < js.length; i += 25000) parts[ck + '_' + (n++)] = js.slice(i, i + 25000);   // 한 칸 100KB 제한 (한글 3바이트)
    if (n <= 40) { parts[ck] = String(n); CACHE().putAll(parts, 900); }
  } catch (e) { /* 캐시 실패는 무시 */ }
  r.now = fmt_(new Date(), 'yyyy-MM-dd HH:mm:ss');
  return r;
}

/** 사진 1장 업로드 → 해당 점검항목 사진 폴더에 고유 이름으로 저장 */
function uploadPhoto(dataUrl, meta) {
  meta = meta || {};
  const user = needUser_(meta.user);
  const m = String(dataUrl).match(/^data:(image\/[a-zA-Z+.-]+);base64,(.+)$/);
  if (!m) throw new Error('이미지 형식 오류');
  const key = String(meta.key || Utilities.getUuid());
  const role = meta.role === 'orig' || meta.role === 'full' ? meta.role : '';
  const folder = role ? markFolder_(meta.code, meta.insp) : itemPhotoFolder_(meta.code, meta.insp);
  const name = `${role ? (role === 'orig' ? '원본_' : '표시본_') : ''}업로드중_${fmt_(new Date(), 'yyyyMMdd_HHmmss')}_${safeName_(user, 12)}_${key.replace(/-/g, '').slice(0, 8)}.jpg`;   // 저장 시 「캡션_번호」로 바뀜
  const file = folder.createFile(Utilities.newBlob(Utilities.base64Decode(m[2]), m[1], name));
  file.setDescription(`key:${key} hash:${meta.hash || ''} by:${user} at:${fmt_(new Date(), 'yyyy-MM-dd HH:mm:ss')}${role ? ' role:' + role : ''}`);
  return { fileId: file.getId(), name };
}

/** 저장하지 않고 버린 업로드(삭제·다시 촬영·취소) → 휴지통 + 이력 */
function discardUpload(fileId, key, user) {
  user = needUser_(user);
  return withLock_(() => {
    try {
      const f = DriveApp.getFileById(fileId);
      if (String(f.getDescription() || '').indexOf('key:' + key) !== 0) return { ok: false };
      f.moveTo(trashFolder_());
      log_(user, '업로드취소', { id: '', code: '', insp: '' }, f.getName(), fileId);
      return { ok: true };
    } catch (e) { return { ok: false }; }
  });
}

/** 항목 저장 (새로 만들기 / 수정). 수정은 버전이 맞아야 함 */
function saveItem(it, user) {
  user = needUser_(user);
  return withLock_(() => {
    const now = fmt_(new Date(), 'yyyy-MM-dd HH:mm:ss');
    let row = 0, old = null;
    // 오프라인 대기분 재전송: 휴대폰이 정한 새 ID로 이미 저장돼 있으면 같은 항목을 고침 (중복 생성 방지)
    if (!it.id && it.newId && /^[\w-]{8,64}$/.test(String(it.newId))) {
      const g0 = getItem_(it.newId);
      if (g0 && g0.item.by === user && !g0.item.del) { it.id = it.newId; it.ver = g0.item.ver; }
      else if (g0) throw new Error('DELETED:이미 삭제된 항목입니다.');
    }
    if (it.id) {
      const g = getItem_(it.id);
      if (!g || g.item.del) throw new Error('DELETED:다른 사용자가 이미 삭제한 항목입니다.');
      if (+it.ver !== +g.item.ver) throw new Error(`CONFLICT:${g.item.mby}님이 먼저 수정했습니다 (${g.item.mat}).`);
      row = g.row; old = g.item;
    }
    const o = old ? Object.assign({}, old) : {
      id: it.newId && /^[\w-]{8,64}$/.test(String(it.newId)) ? String(it.newId) : Utilities.getUuid(), code: it.code, insp: it.insp, ord: nextOrd_(it.code, it.insp),
      by: user, at: now, ver: 0, del: '', sid: it.sid || ''
    };
    EDIT_KEYS.forEach(k => { o[k] = it[k] || ''; });
    const sent = s => (s in it) ? it[s] : (EXTRA_SLOTS.indexOf(s) >= 0 && (o.kind === '지적' || o.kind === '이행') ? null : undefined);
    SLOTS.forEach(s => { const p = sent(s); if (p !== undefined) setSlot_(o, s, p, old); });
    if (old) {                                                   // 새 항목 어디에도 없는 옛 사진만 휴지통 (칸 이동 고려)
      const keep = {}; slotIds_(o).forEach(f => { keep[f] = 1; });
      slotIds_(old).forEach(f => { if (!keep[f]) trashFile_(f); });
    }
    setMark_(o, it, old);
    o.mby = user; o.mat = now; o.ver = (+o.ver || 0) + 1;
    assignNo_(o, old);
    putItem_(row, o);
    nameFiles_(o);
    const tm = {}; SLOTS.forEach(s => { tm[s] = sent(s); });
    saveThumbs_(o.id, tm, !old, old);
    log_(user, old ? '수정' : '등록', o, summary_(o), slotIds_(o).join(','));
    touch_(o.code);
    return o;
  });
}

/** 저장 후에 끝난 업로드를 항목에 연결 */
function setItemPhoto(itemId, slot, key, photo, user) {
  user = needUser_(user);
  return withLock_(() => {
    const g = getItem_(itemId);
    if (g && g.item[slot + 'Key'] !== key) slot = SLOTS.find(s => g.item[s + 'Key'] === key) || slot;   // 칸이 옮겨졌으면 따라감

/** 순서 바꾸기 */
function moveItem(id, dir, user) {
  needUser_(user);
  return withLock_(() => {
    const g = getItem_(id);
    if (!g || g.item.del) return { ok: false };
    const list = readTab_('items').map(rowToItem_)
      .filter(o => o.code === g.item.code && o.insp === g.item.insp && !o.del)
      .sort((a, b) => a.ord - b.ord);
    const i = list.findIndex(o => o.id === id), j = i + (dir < 0 ? -1 : 1);
    if (i < 0 || j < 0 || j >= list.length) return { ok: false };
    const a = getItem_(list[i].id), b = getItem_(list[j].id);
    const t = a.item.ord; a.item.ord = b.item.ord; b.item.ord = t;
    if (a.item.ord === b.item.ord) b.item.ord = a.item.ord + (dir < 0 ? -1 : 1);
    putItem_(a.row, a.item); putItem_(b.row, b.item);
    touch_(g.item.code);
    return { ok: true };
  });
}

/** 조사 5종은 초기점검 · 건축물(구조체) · 1·2종 시설물 점검에만 해당 */
const SURVEY_TYPES = /^(초기점검|구조체 점검|[12]종 시설물)$/;
const SURVEY_ITEMS = /^(초기점검|건축물\s*\d+\s*차|[12]종\s*\d+\s*차)$/;
function surveyAllowed_(code, insp) {
  if (SURVEY_ITEMS.test(String(insp || '').trim())) return true;
  const r = readTab_('insp').find(x => x[0] === code && x[2] === insp);
  return !!(r && SURVEY_TYPES.test(r[3]));
}

function rowToSurvey_(r) {
  const s = {};
  SC.forEach((k, i) => { s[k] = r[i] == null ? '' : String(r[i]); });
  const j = (v, d) => { try { return v ? JSON.parse(v) : d; } catch (e) { return d; } };
  s.fields = j(s.fields, {}); s.rows = j(s.rows, []); s.photos = j(s.photos, []);
  s.ord = +s.ord || 0; s.ver = +s.ver || 0;
  return s;
}
function surveyToRow_(s) {
  return SC.map(k => ['fields', 'rows', 'photos'].includes(k) ? JSON.stringify(s[k] || (k === 'fields' ? {} : []))
    : (s[k] == null ? '' : String(s[k])));
}
function getSurvey_(id) {
  const sh = tab_('surveys'), r = findRow_(sh, id);
  if (!r) return null;
  return { row: r, s: rowToSurvey_(sh.getRange(r, 1, 1, SC.length).getDisplayValues()[0]) };
}
function putSurvey_(row, s) {
  if (row) tab_('surveys').getRange(row, 1, 1, SC.length).setNumberFormat('@').setValues([surveyToRow_(s)]);
  else appendRow_('surveys', surveyToRow_(s));
}

/** 조사 저장 (새로 / 수정 — 수정은 버전 확인) */
function saveSurvey(sv, user) {
  user = needUser_(user);
  if (!SURVEY_DEFS.some(d => d.key === sv.type)) throw new Error('조사 종류가 올바르지 않습니다.');
  if (!surveyAllowed_(sv.code, sv.insp)) throw new Error(`${sv.insp}: 조사는 초기점검 · 건축물 · 1·2종 시설물 점검에서만 입력합니다.`);
  return withLock_(() => {
    const now = fmt_(new Date(), 'yyyy-MM-dd HH:mm:ss');
    const g = sv.id ? getSurvey_(sv.id) : null;
    if (g && sv.isNew && g.s.by === user && !g.s.del) sv.ver = g.s.ver;      // 오프라인 대기분 재전송
    if (g && g.s.del) throw new Error('DELETED:다른 사용자가 이미 삭제한 조사입니다.');
    if (g && +sv.ver !== +g.s.ver) throw new Error(`CONFLICT:${g.s.mby}님이 먼저 수정했습니다 (${g.s.mat}).`);
    const s = g ? Object.assign({}, g.s) : {
      id: /^[\w-]{8,64}$/.test(String(sv.id || '')) ? String(sv.id) : Utilities.getUuid(),
      code: sv.code, insp: sv.insp, type: sv.type, ord: nextSurveyOrd_(sv.code, sv.insp, sv.type),
      by: user, at: now, ver: 0, del: ''
    };
    const fields = {};
    Object.keys(sv.fields || {}).forEach(k => { fields[k] = String(sv.fields[k] == null ? '' : sv.fields[k]).slice(0, 2000); });
    s.fields = fields;
    s.location = fields.location || '';
    s.rows = (sv.rows || []).slice(0, 200).map(r => {
      const o = {};
      Object.keys(r || {}).forEach(k => { o[k] = String(r[k] == null ? '' : r[k]).slice(0, 200); });
      return o;
    }).filter(r => Object.keys(r).some(k => r[k] !== ''));
    s.photos = (sv.photos || []).filter(Boolean).slice(0, 60);
    s.mby = user; s.mat = now; s.ver = (+s.ver || 0) + 1;
    putSurvey_(g ? g.row : 0, s);
    log_(user, g ? '조사수정' : '조사등록', { id: s.id, code: s.code, insp: s.insp }, `${surveyName_(s.type)} · ${s.location}`, '');
    touch_(s.code);
    return s;
  });
}

function moveSurvey(id, dir, user) {

    if (i < 0 || j < 0 || j >= list.length) return { ok: false };
    const a = getSurvey_(list[i].id), b = getSurvey_(list[j].id);
    const t = a.s.ord; a.s.ord = b.s.ord; b.s.ord = t;
    if (a.s.ord === b.s.ord) b.s.ord = a.s.ord + (dir < 0 ? -1 : 1);
    putSurvey_(a.row, a.s); putSurvey_(b.row, b.s);
    touch_(g.s.code);
    return { ok: true };
  });
}

/** 계산 가능한 값만 (입력값이 없으면 빈칸) */
function surveyCalc_(def, s) {
  const num = v => { const n = parseFloat(String(v == null ? '' : v).replace(/,/g, '')); return isFinite(n) ? n : null; };
  const f = s.fields || {}, rows = s.rows || [];
  const r1 = x => Math.round(x * 10) / 10;
  const avg = a => a.reduce((x, y) => x + y, 0) / a.length;
  const per = rows.map(() => []);
  const sum = [];
  if (def.key === 'concrete') {
    const v = rows.map(r => num(r.v)).filter(x => x !== null);
    if (v.length) {
      const unit = f.unit || 'MPa';
      sum.push(`측정 횟수 ${v.length}회`, `평균값 ${r1(avg(v))} ${unit}`, `최소 ${r1(Math.min(...v))} · 최대 ${r1(Math.max(...v))}`);
      const fck = num(f.fck);
      if (fck && unit === 'MPa') sum.push(`설계기준강도 대비 ${Math.round(avg(v) / fck * 100)}%`);
    }
  } else if (def.key === 'rebar') {
    const sp = rows.map(r => num(r.spacing)).filter(x => x !== null);
    const cv = rows.map(r => num(r.cover)).filter(x => x !== null);
    if (rows.length) sum.push(`측정 ${rows.length}개소`);
    if (sp.length) sum.push(`평균 간격 ${r1(avg(sp))} mm`);
    if (cv.length) sum.push(`평균 피복 ${r1(avg(cv))} mm (최소 ${r1(Math.min(...cv))})`);
  } else if (def.key === 'defect') {
    const c = {};
    rows.forEach(r => { const k = r.dtype || '기타'; c[k] = (c[k] || 0) + 1; });
    if (rows.length) sum.push(Object.keys(c).map(k => `${k} ${c[k]}건`).join(', '));
  } else if (def.key === 'tilt') {
    let worst = null;
    rows.forEach((r, i) => {
      const h = num(r.h), d = num(r.d);
      if (h && d) {
        const k = Math.abs(h / d);
        per[i] = [`1/${Math.round(k)}`];
        if (worst === null || k < worst) worst = k;
      } else per[i] = [''];
    });
    if (worst !== null) sum.push(`최대 기울기 1/${Math.round(worst)}`);
  } else if (def.key === 'settle') {
    const base = num(f.refh);
    const hs = rows.map(r => num(r.h));
    rows.forEach((r, i) => { per[i] = [hs[i] !== null && base !== null ? String(r1(hs[i] - base)) : '']; });
    const v = hs.filter(x => x !== null);
    if (v.length >= 2) sum.push(`측정 지점 간 최대 높이 차 ${r1(Math.max(...v) - Math.min(...v))} mm`);
    if (base !== null && v.length) sum.push(`기준점 대비 최대 ${r1(Math.max(...v.map(x => Math.abs(x - base))))} mm`);
  }
  return { per, summary: sum.join('  ·  ') };
}

/** 작은 사진(목록용) */
function getThumbs(ids) {
  ids = (ids || []).slice(0, 60);
  const out = {};
  let hit = {};
  try { hit = CACHE().getAll(ids.map(id => 'TH_' + id)); } catch (e) { hit = {}; }
  const miss = [];
  ids.forEach(id => {
    const c = hit['TH_' + id];
    if (c) { try { out[id] = JSON.parse(c); return; } catch (e) { /* 아래로 */ } }
    miss.push(id);
  });
  if (miss.length) {
    const sh = tab_('thumbs'), last = sh.getLastRow();
    if (last >= 2) {
      const want = {};
      miss.forEach(id => { want[id] = 1; });
      const col = sh.getRange(2, 1, last - 1, 1).getDisplayValues();       // ID 열만 한 번 읽고
      const rows = [];
      col.forEach((v, i) => { if (want[v[0]]) rows.push(i + 2); });
      const put = {};
      if (rows.length > 8) {                                               // 많으면 표 전체를 한 번에
        sh.getRange(2, 1, last - 1, TH_N).getDisplayValues().forEach(v => {
          if (want[v[0]]) { out[v[0]] = thRow_(v); put['TH_' + v[0]] = JSON.stringify(out[v[0]]); }
        });
      } else {
        rows.forEach(r => {
          const v = sh.getRange(r, 1, 1, TH_N).getDisplayValues()[0];
          out[v[0]] = thRow_(v); put['TH_' + v[0]] = JSON.stringify(out[v[0]]);
        });
      }
      try { CACHE().putAll(put, 21600); } catch (e) { /* 무시 */ }
    }
  }
  return out;
}

function buildInspection(p) {
  if (!p || !p.code || !p.item) throw new Error('용역 또는 점검항목 정보가 없습니다.');
  const allLive = readTab_('items').map(rowToItem_).filter(o => o.code === p.code && o.insp === p.item && !o.del);
  const byId = {};
  allLive.forEach(o => { byId[o.id] = o; });
  // 조사: 순서·포함 설정에 따라, 입력이 있는 조사만
  const cfg = p.surveys || {};
  const order = (cfg.order && cfg.order.length ? cfg.order : SURVEY_DEFS.map(d => d.key))
    .concat(SURVEY_DEFS.map(d => d.key)).filter((k, i, a) => a.indexOf(k) === i);
  const off = cfg.off || {};
  const svAll = surveyAllowed_(p.code, p.item)
    ? readTab_('surveys').map(rowToSurvey_).filter(s => s.code === p.code && s.insp === p.item && !s.del) : [];
  const svGroups = order.filter(k => !off[k]).map(k => ({
    def: SURVEY_DEFS.find(d => d.key === k),
    recs: svAll.filter(s => s.type === k).sort((a, b) => a.ord - b.ord)
  })).filter(g => g.def && g.recs.length);
  const usedIds = {};
  svGroups.forEach(g => g.recs.forEach(s => s.photos.forEach(id => { usedIds[id] = 1; })));

  const live = allLive.filter(o => o.kind !== '조사' && o.kind !== '실측' && o.kind !== '이행').sort((a, b) => a.ord - b.ord);
  const pending = allLive.filter(o => ((o.kind !== '조사' && o.kind !== '실측') || usedIds[o.id]) && SLOTS.some(s => o[s + 'St'] === '업로드중'));
  // 이 점검항목에 연결된 제원 실측 → 사진대지에 '제원 비교' 시트로
  const specMap = {};
  readRecs_('specs').forEach(x => { specMap[x.id] = x; });
  const mlist = readRecs_('measures').filter(x => x.code === p.code && x.insp === p.item && !x.del).sort((a, b) => (a.date + a.at < b.date + b.at ? -1 : 1));
  const allById = {};
  readTab_('items').map(rowToItem_).filter(o => o.code === p.code && !o.del).forEach(o => { allById[o.id] = o; });
  // 이전 차수 지적사항 이행 확인: 휴대폰이 고른 이전 지적 ID 목록 + 이번 점검의 확인 기록
  const fuRows = (p.prev || []).map(x => {
    const o = allById[x.id]; if (!o) return null;
    const fu = allLive.filter(f => f.kind === '이행' && f.sid === o.id).sort((a, b) => a.ord - b.ord).pop() || null;
    return { o, fu, from: o.insp };
  }).filter(Boolean);
  if (pending.length) throw new Error(`${p.item}: 업로드 중인 사진이 ${pending.length}건 있습니다 (${[...new Set(pending.map(o => o.by))].join(', ')}). 끝난 뒤 다시 만들어 주세요.`);
  const ph = (o, s) => o[s + 'Id'] ? { fileId: o[s + 'Id'], w: +o[s + 'W'] || 0, h: +o[s + 'H'] || 0 } : null;
  const items = live.map(o => ({
    kind: o.kind, floor: o.floor, location: o.location, caption: o.caption, issue: o.issue,
    action: o.action, status: o.status, risk: o.risk, by: o.by, before: ph(o, 'b'), after: ph(o, 'a'),
    befores: slotsOf_(o, 'b').map(t => ph(o, t)), afters: slotsOf_(o, 'a').map(t => ph(o, t))
  }));
  const photos = items.filter(it => it.kind === '현황');
  const defects = items.filter(it => it.kind === '지적');
  const ck = (p.checklist && p.checklist.rows && p.checklist.rows.length) ? p.checklist : null;
  if (!items.length && !ck && !svGroups.length && !mlist.length && !fuRows.some(r => r.fu)) throw new Error(p.item + ': 사진·체크리스트·조사·실측 내용이 없습니다.');
  p.date = p.date || fmt_(new Date(), 'yyyy-MM-dd');

  const itemFolder = itemFolder_(p.code, p.item);
  const photoFolder = itemPhotoFolder_(p.code, p.item);
  const baseName = safeName_(`${p.date}_${p.item}_${p.name}`, 90) + '_' + fmt_(new Date(), 'HHmmss');

  const ss = SpreadsheetApp.create(baseName);
  const ssFile = DriveApp.getFileById(ss.getId());
  ssFile.moveTo(itemFolder);
  IMG_JOBS = [];
  const first = ss.getSheets()[0];
  first.setName('점검사진');
  writeGridSheet_(first, p, photos, photoFolder);
  if (defects.length) writeDefectSheet_(ss.insertSheet('지적사항'), p, defects, photoFolder);
  if (fuRows.length) writeFollowSheet_(ss.insertSheet('이전 지적 이행확인'), p, fuRows, ph);
  mlist.forEach((m, i) => writeCompareSheet_(ss.insertSheet(safeName_(`제원비교${mlist.length > 1 ? i + 1 : ''} ${(specMap[m.specId] || {}).name || ''}`, 28)),
    p, m, specMap[m.specId] || {}, allById));
  svGroups.forEach(g => writeSurveySheet_(ss.insertSheet(g.def.name.replace(/\s*조사$/, '') + ' 조사'), p, g.def, g.recs, byId, photoFolder));
  if (ck) writeChecklistSheet_(ss.insertSheet('체크리스트'), p, ck);
  writeListSheet_(ss.insertSheet('점검목록'), p, items);
  ss.setActiveSheet(first);
  const jobs = IMG_JOBS; IMG_JOBS = null;
  const cnt = { photo: photos.length, defect: defects.length, open: defects.filter(d => d.status !== '조치완료').length,
    survey: svGroups.reduce((a, g) => a + g.recs.length, 0), measure: mlist.length,
    follow: fuRows.length, followDone: fuRows.filter(r => r.fu && r.fu.status === '이행완료').length };
  const st = { id: Utilities.getUuid(), ssId: ss.getId(), baseName, folderId: itemFolder.getId(), p, cnt, jobs, i: 0 };
  return runBuild_(st, p.budgetMs);
}

/** 사진대지: 사진 넣기 이어서 (휴대폰이 남은 게 없을 때까지 부름) */
function buildContinue(id, budgetMs) {
  const f = buildStateFile_(id);
  if (!f) throw new Error('사진대지 작업 정보를 찾을 수 없습니다. 처음부터 다시 만들어 주세요.');
  return runBuild_(JSON.parse(f.getBlob().getDataAsString()), budgetMs, f);
}
const BUILD_BUDGET_MS = 240000;                      // 한 번에 4분까지 (Apps Script 6분 제한 안쪽)
function buildStateFile_(id) {
  const it = cachedFolder_('FD_JOBS', () => getOrCreateSub_(getRootFolder_(), '_작업중')).getFilesByName('build_' + id + '.json');
  return it.hasNext() ? it.next() : null;
}
function runBuild_(st, budgetMs, stateFile) {
  const t0 = Date.now(), budget = Math.min(BUILD_BUDGET_MS, +budgetMs || BUILD_BUDGET_MS);
  const ss = SpreadsheetApp.openById(st.ssId), shs = {};
  while (st.i < st.jobs.length && Date.now() - t0 < budget) {
    const j = st.jobs[st.i];
    const sh = shs[j.s] || (shs[j.s] = ss.getSheetByName(j.s));
    if (sh) placeNow_(sh, { fileId: j.f, w: j.w, h: j.h }, j.r, j.c, j.W, j.bw, j.bh);
    st.i++;
  }
  if (st.i < st.jobs.length) {                        // 남았으면 상태를 저장하고 휴대폰이 다시 부름
    const body = JSON.stringify(st);
    if (stateFile) stateFile.setContent(body);
    else cachedFolder_('FD_JOBS', () => getOrCreateSub_(getRootFolder_(), '_작업중')).createFile('build_' + st.id + '.json', body, 'application/json');
    return { pending: st.id, item: st.p.item, done: st.i, total: st.jobs.length };
  }
  if (stateFile) { try { stateFile.setTrashed(true); } catch (e) { /* 무시 */ } }
  return finishBuild_(st, ss);
}
function finishBuild_(st, ss) {
  const p = st.p, baseName = st.baseName, itemFolder = DriveApp.getFolderById(st.folderId), cnt = st.cnt;
  const ssFile = DriveApp.getFileById(ss.getId());
  SpreadsheetApp.flush();

  const headers = { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() };
  const base = `https://docs.google.com/spreadsheets/d/${ss.getId()}/export`;
  const xlsxFile = itemFolder.createFile(
    UrlFetchApp.fetch(base + '?format=xlsx', { headers }).getBlob().setName(baseName + '.xlsx'));
  let pdfFile = null;
  if (CONFIG.MAKE_PDF) {
    const q = ['format=pdf', 'size=A4', 'portrait=true', 'fitw=true', 'gridlines=false', 'printtitle=false',
      'sheetnames=false', 'pagenum=CENTER', 'fzr=false', 'top_margin=0.5', 'bottom_margin=0.5',
      'left_margin=0.4', 'right_margin=0.4', 'horizontal_alignment=CENTER'].join('&');
    pdfFile = itemFolder.createFile(
      UrlFetchApp.fetch(base + '?' + q, { headers }).getBlob().setName(baseName + '.pdf'));
  }
  if (!CONFIG.KEEP_GSHEET) ssFile.setTrashed(true);
  if (CONFIG.SHARE_ANYONE_WITH_LINK) {
    [xlsxFile, pdfFile].filter(Boolean).forEach(f =>
      f.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW));
  }

  const links = {
    xlsx: xlsxFile.getUrl(), download: 'https://drive.google.com/uc?export=download&id=' + xlsxFile.getId(),
    pdf: pdfFile ? pdfFile.getUrl() : '', folder: itemFolder.getUrl()
  };
  withLock_(() => saveResult_(p, links, cnt));
  return { item: p.item, name: baseName, count: cnt, photos: st.jobs.length, xlsxUrl: links.xlsx, xlsxDownload: links.download, pdfUrl: links.pdf, folderUrl: links.folder };
}

/* ======================= 사진대지 시트 ======================= */

const GRID_W = [340, 340];
const GRID_PH = 235;
const GRID_CAP = 26;
const DEF_W = [90, 330, 330];
const DEF_PH = 230;
const STATUS_COLOR = { '양호': '#d9f2e3', '지적': '#fbdcdc', '조치완료': '#dbe8fb' };
const CK_COLOR = { '부적정': '#fbdcdc', '미시행': '#fff1d6' };
const BORDER = ['#555555', SpreadsheetApp.BorderStyle.SOLID];

function roundLabel_(p) {
  if (/초기점검/.test(p.item)) return '초기점검';
  const m = String(p.item).match(/(\d+)\s*차$/);
  const base = m ? `제${m[1]}차 정기안전점검` : p.item;
  return p.tag ? `${base} (${p.tag})` : base;
}

function writeHeader_(sh, p, title, widths) {
  const n = widths.length;
  if (sh.getMaxColumns() > n) sh.deleteColumns(n + 1, sh.getMaxColumns() - n);
  widths.forEach((w, i) => sh.setColumnWidth(i + 1, w));
  sh.setHiddenGridlines(true);
  sh.getRange(1, 1, sh.getMaxRows(), n).setFontSize(10).setVerticalAlignment('middle');
  sh.getRange(1, 1, 1, n).merge();
  sh.getRange(1, 1).setValue(title).setFontSize(15).setFontWeight('bold').setHorizontalAlignment('center');
  sh.setRowHeight(1, 40);
  const info = [
    ['용 역 명', p.name],
    ['점검항목', `${p.item}  ·  ${roundLabel_(p)}`],
    ['점검일 / 점검자', `${p.date}  /  ${p.inspector || ''}`],
    ['점검기관 / 책임기술자', `${CONFIG.ORG_NAME}  /  ${CONFIG.CHIEF_ENGINEER || ''}`]
  ];
  if (p.timing) info.push(['점검시기 기준', p.timing]);
  info.forEach((r, i) => {
    const row = 2 + i;
    sh.getRange(row, 1, 1, n).merge();
    sh.getRange(row, 1).setValue(txt_(`${r[0]} : ${r[1] || ''}`)).setHorizontalAlignment('left');
    sh.setRowHeight(row, 24);
  });
  sh.getRange(2, 1, info.length, n).setBorder(true, true, true, true, false, true, ...BORDER).setBackground('#f5f7fa');
  sh.setRowHeight(2 + info.length, 10);
  return 3 + info.length;
}

function ensureRows_(sh, need) {
  if (sh.getMaxRows() < need) sh.insertRowsAfter(sh.getMaxRows(), need - sh.getMaxRows());
}

function writeGridSheet_(sh, p, list, photoFolder) {
  ensureRows_(sh, 10 + Math.ceil(list.length / 2) * 2);
  let r = writeHeader_(sh, p, `${p.type || p.item} 정기안전점검 사진`, GRID_W);
  sh.getRange(r, 1, 1, 2).merge();
  sh.getRange(r, 1).setValue(`${roundLabel_(p)} 사진`).setFontWeight('bold')
    .setHorizontalAlignment('center').setBackground('#dfe6ee');
  sh.getRange(r, 1, 1, 2).setBorder(true, true, true, true, null, null, ...BORDER);
  sh.setRowHeight(r, 28);
  r++;
  if (!list.length) {
    sh.getRange(r, 1, 1, 2).merge();
    sh.getRange(r, 1).setValue('(현황 사진 없음)').setHorizontalAlignment('center').setFontColor('#999999');
    return;
  }
  for (let i = 0; i < list.length; i += 2) {
    sh.setRowHeight(r, GRID_PH);
    sh.setRowHeight(r + 1, GRID_CAP);
    for (let j = 0; j < 2; j++) {
      const it = list[i + j];
      if (!it) break;
      const no = i + j + 1, col = j + 1;
      placeImage_(sh, it.before, r, col, GRID_W, GRID_W[j], GRID_PH);
      sh.getRange(r + 1, col).setValue(txt_(`<사진${no}> ${captionOf_(it)}`))
        .setHorizontalAlignment('center').setWrap(true).setFontSize(9);
    }
    sh.getRange(r, 1, 2, 2).setBorder(true, true, true, true, true, true, ...BORDER);
    r += 2;
  }
}

function writeDefectSheet_(sh, p, list, photoFolder) {
  ensureRows_(sh, 10 + list.reduce((a, it) => a + 6 + photoRows_(it.befores, it.afters), 0));
  let r = writeHeader_(sh, p, `${p.type || p.item} 지적사항 및 조치결과`, DEF_W);
  list.forEach((it, idx) => {
    r = writePairBlock_(sh, r, '지적 ' + (idx + 1), it.befores, it.afters, ['조치 전', '조치 후', '현장 사진'],
      ['위    치', '지적사항', '조치사항', '상태 / 위험도'],
      [join_([it.floor, it.location], ' '), it.issue, it.action, join_([it.status, it.risk ? '위험도 ' + it.risk : ''], '  /  ')],
      STATUS_COLOR[it.status], 3, '(사진 없음)');
  });
}
/** 사진 줄 수: 조치 후가 있으면 「조치 전 | 조치 후」 한 줄에 한 쌍, 없으면 1장은 크게 · 여러 장은 2장씩 */
function photoRows_(L, R) {
  L = L || []; R = R || [];
  if (R.length) return Math.max(L.length, R.length, 1);
  return L.length <= 1 ? 1 : Math.ceil(L.length / 2);
}
const PH_H = n => n <= 1 ? DEF_PH : 200;
/** 사진 묶음 + 정보 표 (지적사항 · 이행확인 공용). colorRow: 색칠할 정보 줄 번호(0부터) */
function writePairBlock_(sh, r, title, L, R, heads, labels, vals, color, colorRow, emptyR, forcePair) {
  L = (L || []).filter(Boolean); R = (R || []).filter(Boolean);
  const pair = R.length > 0 || !!forcePair, n = pair ? Math.max(L.length, R.length, 1) : photoRows_(L, R), h = PH_H(n);
  sh.setRowHeight(r, 26);
  sh.getRange(r, 1).setValue(title);
  if (pair) { sh.getRange(r, 2).setValue(heads[0]); sh.getRange(r, 3).setValue(heads[1]); }
  else { sh.getRange(r, 2, 1, 2).merge(); sh.getRange(r, 2).setValue(heads[2] + (L.length > 1 ? ` (${L.length}장)` : '')); }
  sh.getRange(r, 1, 1, 3).setBackground('#dfe6ee').setFontWeight('bold').setHorizontalAlignment('center');
  for (let i = 0; i < n; i++) {
    const row = r + 1 + i;
    sh.setRowHeight(row, h);
    if (i === 0) sh.getRange(row, 1).setValue('사 진').setBackground('#f5f7fa').setFontWeight('bold').setHorizontalAlignment('center');
    if (pair) {
      if (L[i]) placeImage_(sh, L[i], row, 2, DEF_W, DEF_W[1], h);

      else if (i === 0 && typeof emptyR === 'string') sh.getRange(row, 3).setValue(emptyR).setHorizontalAlignment('center').setFontColor('#999999');
    } else if (L.length <= 1) {
      sh.getRange(row, 2, 1, 2).merge();
      placeImage_(sh, L[0] || null, row, 2, DEF_W, DEF_W[1] + DEF_W[2], h);
    } else {
      if (L[i * 2]) placeImage_(sh, L[i * 2], row, 2, DEF_W, DEF_W[1], h);
      if (L[i * 2 + 1]) placeImage_(sh, L[i * 2 + 1], row, 3, DEF_W, DEF_W[2], h);
    }
  }
  if (n > 1) sh.getRange(r + 1, 1, n, 1).merge().setVerticalAlignment('middle');
  const base = r + 1 + n;
  labels.forEach((lb, i) => {
    const row = base + i;
    sh.setRowHeight(row, /사항|내용/.test(lb) ? 48 : 26);
    sh.getRange(row, 1).setValue(lb).setBackground('#f5f7fa').setFontWeight('bold').setHorizontalAlignment('center').setWrap(true);
    sh.getRange(row, 2, 1, 2).merge().setWrap(true).setHorizontalAlignment('left');
    sh.getRange(row, 2).setValue(txt_(vals[i]));
  });
  if (color && colorRow != null) sh.getRange(base + colorRow, 2, 1, 2).setBackground(color);
  sh.getRange(r, 1, 1 + n + labels.length, 3).setBorder(true, true, true, true, true, true, ...BORDER);
  sh.setRowHeight(base + labels.length, 14);
  return base + labels.length + 1;
}

/* 사진 넣기는 모아 두었다가 시간 안에서 차례로 (사진이 수백 장이면 여러 번에 나눠서) */
let IMG_JOBS = null;
function placeImage_(sh, ph, row, col, widths, boxW, boxH) {
  if (!ph || !ph.fileId) {
    sh.getRange(row, col).setValue('(사진 없음)').setHorizontalAlignment('center').setFontColor('#999999');
    return;
  }
  if (IMG_JOBS) { IMG_JOBS.push({ s: sh.getName(), f: ph.fileId, w: +ph.w || 0, h: +ph.h || 0, r: row, c: col, W: widths, bw: boxW, bh: boxH }); return; }
  placeNow_(sh, ph, row, col, widths, boxW, boxH);
}
function placeNow_(sh, ph, row, col, widths, boxW, boxH) {
  let blob;
  try { blob = DriveApp.getFileById(ph.fileId).getBlob(); }
  catch (e) { sh.getRange(row, col).setValue('(사진 파일 없음)').setHorizontalAlignment('center').setFontColor('#c0392b'); return; }
  const pad = 6;
  const iw = ph.w || 1100, ih = ph.h || 825;
  const s = Math.min((boxW - pad * 2) / iw, (boxH - pad * 2) / ih);
  const w = Math.round(iw * s), h = Math.round(ih * s);
  let c = col, offX = Math.round((boxW - w) / 2);
  while (c - 1 < widths.length && offX >= widths[c - 1]) { offX -= widths[c - 1]; c++; }
  const img = sh.insertImage(blob, c, row, offX, Math.round((boxH - h) / 2));
  img.setWidth(w).setHeight(h);
}

/* ---- 파일 이름: 「캡션_01.jpg」 — 번호는 잠금 안에서 한 명씩 차례로, 지운 번호는 다시 쓰지 않음 ---- */
function baseOf_(o) {
  if (o.kind === '실측') return String(o.caption || '실측').replace(/[\\/\r\n\t]/g, '_').trim().slice(0, 80);   // 규격의 * 는 그대로 (예: ㅁ-125*100*10)
  const b = o.kind === '지적' || o.kind === '이행' ? o.issue : o.caption;
  return safeName_(String(b || o.kind || '사진').trim(), 50);
}
/** 처음 저장할 때 또는 캡션이 바뀌었을 때만 새 번호 (잠금 안에서 호출) */
function assignNo_(o, old) {
  const base = baseOf_(o);
  if (o.no && old && baseOf_(old) === base) return;
  if (o.no && !old) return;
  o.no = nextNo_(o.code + '|' + o.insp + '|' + base);
}
function nextNo_(key) {
  const sh = tab_('counters');
  const r = findRow_(sh, key);
  const n = (r ? +sh.getRange(r, 2).getDisplayValues()[0][0] || 0 : 0) + 1;
  if (r) sh.getRange(r, 2).setNumberFormat('@').setValue(String(n));
  else appendRow_('counters', [key, String(n)]);
  return n;
}
function alpha_(n) { let s = ''; n = Math.max(1, n | 0); while (n > 0) { n--; s = String.fromCharCode(97 + n % 26) + s; n = Math.floor(n / 26); } return s; }
function fileName_(o, slot) {
  const n = String(slot || '').slice(1), sub = n ? '_' + n : '';          // 두 번째 사진부터 _2, _3, _4
  const tail = o.kind === '지적' ? (slot[0] === 'a' ? '_조치후' : '_조치전') + sub : o.kind === '이행' ? '_이행확인' + sub : '';
  return `${baseOf_(o)}_${pad2_(+o.no || 0)}${tail}.jpg`;
}
function nameFiles_(o) {
  SLOTS.forEach(s => {
    const id = o[s + 'Id'];
    if (id && !isPending_(id) && o.no) { try { DriveApp.getFileById(id).setName(fileName_(o, s)); } catch (e) { /* 없음 */ } }
  });
  if (o.mark && o.no) {
    const m = markOf_(o);
    SLOTS.forEach(s => { const x = m[s]; if (x) ['orig', 'full'].forEach(r => {
      const f = x[r];
      if (f && !isPending_(f)) { try { DriveApp.getFileById(f).setName(fileName_(o, s).replace(/\.jpg$/, r === 'orig' ? '_원본.jpg' : '_표시.jpg')); } catch (e) { /* 없음 */ } }
    }); });
  }
}

function getDb_() {
  if (RT.db) return RT.db;
  const id = P_('DB_ID');
  if (id) {
    try { return (RT.db = SpreadsheetApp.openById(id)); } catch (e) { /* 없어졌으면 새로 */ }
  }
  return (RT.db = withLock_(() => {
    const id2 = PropertiesService.getScriptProperties().getProperty('DB_ID');
    if (id2 && id2 !== id) return SpreadsheetApp.openById(id2);
    const ss = SpreadsheetApp.create('현장앱_DB');
    DriveApp.getFileById(ss.getId()).moveTo(getRootFolder_());
    initTab_(ss.getSheets()[0], 'svc');
    setP_('DB_ID', ss.getId());

function ensureCols_(sh, need) {
  if (sh.getMaxColumns() < need) sh.insertColumnsAfter(sh.getMaxColumns(), need - sh.getMaxColumns());
}

function initTab_(sh, key) {
  const t = DB_TABS[key];
  sh.setName(t.name);
  sh.clear();
  ensureCols_(sh, t.head.length);
  sh.getRange(1, 1, sh.getMaxRows(), t.head.length).setNumberFormat('@');
  sh.getRange(1, 1, 1, t.head.length).setValues([t.head])
    .setFontWeight('bold').setBackground('#16324f').setFontColor('#ffffff');
  sh.setFrozenRows(1);
  return sh;
}

const TAB_CACHE = {};
function tab_(key) {
  if (TAB_CACHE[key]) return TAB_CACHE[key];
  const ss = getDb_();
  let sh = ss.getSheetByName(DB_TABS[key].name);
  if (!sh) sh = withLock_(() => ss.getSheetByName(DB_TABS[key].name) || initTab_(ss.insertSheet(), key));
  const head = DB_TABS[key].head;                                       // 열이 늘어난 버전으로 올라온 경우 열·제목줄 보정 (탭마다 한 번만)
  const sig = head.length + ':' + head[head.length - 1];
  if (P_('HDR_' + key) !== sig) {
    ensureCols_(sh, head.length);
    if (sh.getRange(1, head.length).getDisplayValues()[0][0] !== head[head.length - 1]) {
      sh.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight('bold').setBackground('#16324f').setFontColor('#ffffff');
    }
    setP_('HDR_' + key, sig);
  }
  TAB_CACHE[key] = sh;
  return sh;
}

}

    return a;
  });
  ensureRows_(sh, data.length + 1);
  sh.getRange(2, 1, data.length, n).setNumberFormat('@').setValues(data);
}

/** 한 줄 추가 (잠금 안에서 호출) */
function appendRow_(key, arr) {
  const sh = tab_(key), n = DB_TABS[key].head.length;
  const r = sh.getLastRow() + 1;
  ensureRows_(sh, r);
  const a = arr.slice(0, n).map(v => v == null ? '' : String(v));
  while (a.length < n) a.push('');
  sh.getRange(r, 1, 1, n).setNumberFormat('@').setValues([a]);
  return r;
}

function findRow_(sh, id) {
  if (!id) return 0;
  const last = sh.getLastRow();
  if (last < 2) return 0;
  const f = sh.getRange(2, 1, last - 1, 1).createTextFinder(String(id)).matchEntireCell(true).findNext();
  return f ? f.getRow() : 0;
}

function rowToItem_(r) {
  const o = {};
  IC.forEach((k, i) => { o[k] = r[i] == null ? '' : String(r[i]); });
  o.ord = +o.ord || 0;
  o.ver = +o.ver || 0;
  let x = {};
  try { x = o.ext ? JSON.parse(o.ext) : {}; } catch (e) { x = {}; }
  EXTRA_SLOTS.forEach(s => SF.forEach(f => { o[s + f] = x[s] && x[s][f] != null ? String(x[s][f]) : ''; }));
  delete o.ext;
  return o;
}
function itemToRow_(o) {
  const x = {};
  EXTRA_SLOTS.forEach(s => { if (o[s + 'Id']) { x[s] = {}; SF.forEach(f => { x[s][f] = o[s + f] == null ? '' : String(o[s + f]); }); } });
  const t = Object.assign({}, o, { ext: Object.keys(x).length ? JSON.stringify(x) : '' });
  return IC.map(k => t[k] == null ? '' : String(t[k]));
}

function getItem_(id) {
  const sh = tab_('items');
  const r = findRow_(sh, id);
  if (!r) return null;
  return { row: r, item: rowToItem_(sh.getRange(r, 1, 1, IC.length).getDisplayValues()[0]) };
}

function putItem_(row, o) {
  if (row) tab_('items').getRange(row, 1, 1, IC.length).setNumberFormat('@').setValues([itemToRow_(o)]);
  else appendRow_('items', itemToRow_(o));
}

function nextOrd_(code, insp) {

const isPending_ = id => /^pending:/.test(String(id || ''));

/** 사진 칸 채우기. 같은 업로드(키)가 이미 '완료'로 연결돼 있으면 그대로 유지 */
/* ---- 사진 표시(원·화살표 등): 원본 · 원본 해상도 표시본 · 도형 ---- */
function markOf_(o) { try { return JSON.parse(o.mark || '{}') || {}; } catch (e) { return {}; } }
function markFiles_(o) {
  const m = markOf_(o), out = [];
  SLOTS.forEach(s => { const x = m[s]; if (x) ['orig', 'full'].forEach(r => { if (x[r] && !isPending_(x[r])) out.push(x[r]); }); });
  return out;
}
function markFolder_(code, insp) {
  return cachedFolder_('FD_' + code + '|' + insp + '|원본', () => getOrCreateSub_(itemFolder_(code, insp), '원본'));
}
/** 저장할 때 표시 정보 정리: 같은 key면 올라간 파일 ID 유지, 빠진 파일은 휴지통 */
function setMark_(o, it, old) {
  const om = old ? markOf_(old) : {}, nm = {};
  const prevByKey = k => { const t = SLOTS.find(t2 => old && old[t2 + 'Key'] === k && om[t2]); return t ? om[t] : null; };
  SLOTS.forEach(s => {
    const p = it[s], x = p && p.mark, prev = (p && prevByKey(String(p.key || ''))) || om[s] || {};
    if (!x || !Array.isArray(x.shapes)) {                       // 표시 정보 없이 저장 → 같은 사진이면 기존 표시 유지 (칸이 옮겨져도)
      const pk = p && prevByKey(String(p.key || ''));
      if (pk) nm[s] = pk;
      else if (p === undefined && om[s]) nm[s] = om[s];
      return;
    }
    const y = { w: +x.w || 0, h: +x.h || 0, shapes: x.shapes.slice(0, 80) };
    ['orig', 'full'].forEach(r => {
      const key = String(x[r + 'Key'] || '');
      y[r + 'Key'] = key;
      if (x[r + 'Id']) y[r] = String(x[r + 'Id']);
      else if (key && prev[r + 'Key'] === key && prev[r]) y[r] = prev[r];
      else y[r] = key ? 'pending:' + key : '';
    });
    nm[s] = y;
  });
  const keep = {};
  SLOTS.forEach(s => { if (nm[s]) ['orig', 'full'].forEach(r => { if (nm[s][r]) keep[nm[s][r]] = 1; }); });
  SLOTS.forEach(s => { const x = om[s]; if (x) ['orig', 'full'].forEach(r => { const f = x[r]; if (f && !isPending_(f) && !keep[f]) trashFile_(f); }); });
  o.mark = Object.keys(nm).length ? JSON.stringify(nm) : '';
}
/** 원본·표시본 업로드가 끝나면 항목에 연결 */
function setItemMark(itemId, slot, role, key, fileId, user) {
  user = needUser_(user);
  if (role !== 'orig' && role !== 'full') throw new Error('잘못된 요청');
  return withLock_(() => {
    const g = getItem_(itemId);
    const m = g ? markOf_(g.item) : {};
    if (!(m[slot] && m[slot][role + 'Key'] === key)) slot = SLOTS.find(s => m[s] && m[s][role + 'Key'] === key) || slot;
    const x = m[slot];
    if (!g || g.item.del || !x || x[role + 'Key'] !== key) {
      trashFile_(fileId);
      log_(user, '미사용업로드', { id: itemId, code: '', insp: '' }, role, fileId);
      return { orphan: true };
    }
    if (x[role] === fileId) return { ok: true };
    x[role] = fileId;
    const o = g.item; o.mark = JSON.stringify(m);
    putItem_(g.row, o);
    nameFiles_(o);
    touch_(o.code);
    return { ok: true };
  });
}

function setSlot_(o, s, p, old) {
  const F = ['Id', 'W', 'H', 'Hash', 'Key', 'St'];
  if (!p) { F.forEach(f => { o[s + f] = ''; }); return; }
  const key = String(p.key || '');
  const from = old && key ? SLOTS.find(t => old[t + 'Key'] === key && old[t + 'St'] === '완료') : null;   // 칸이 옮겨져도 같은 사진이면 그대로
  if (from) {
    F.forEach(f => { o[s + f] = old[from + f]; });
    return;
  }
  o[s + 'Key'] = key;
  o[s + 'Hash'] = p.hash || '';
  o[s + 'W'] = p.w || '';
  o[s + 'H'] = p.h || '';
  if (p.fileId) { o[s + 'Id'] = p.fileId; o[s + 'St'] = '완료'; }
  else { o[s + 'Id'] = 'pending:' + key; o[s + 'St'] = '업로드중'; }
}

/** 썸네일 저장: 사진 있음+썸네일 → 저장 / null → 지움 / undefined 또는 썸네일 없음 → 그대로 */
const TH_SLOTS = ['b', 'a', 'b2', 'b3', 'b4', 'a2', 'a3', 'a4'];           // 썸네일 탭 열 순서
const TH_N = TH_SLOTS.length + 1;
const thRow_ = v => { const o = {}; TH_SLOTS.forEach((s, i) => { o[s] = v[i + 1] || ''; }); return o; };
/** 썸네일 저장: 칸별 {사진+썸네일 → 저장 / null → 지움 / undefined → 그대로 / 썸네일 없는 사진 → 같은 사진의 옛 썸네일} */
function saveThumbs_(id, map, isNew, old) {
  const sh = tab_('thumbs');
  const r = isNew ? 0 : findRow_(sh, id);
  const cur = r ? sh.getRange(r, 1, 1, TH_N).getDisplayValues()[0] : [id].concat(TH_SLOTS.map(() => ''));
  const curOf = s => cur[TH_SLOTS.indexOf(s) + 1] || '';
  const byKey = k => { const t = old && k ? SLOTS.find(t2 => old[t2 + 'Key'] === k) : null; return t ? curOf(t) : null; };
  const row = [id].concat(TH_SLOTS.map(s => {
    const p = map[s];
    if (p === undefined) return curOf(s);
    if (p === null || !p) return '';
    if (p.thumb) return p.thumb;
    const k = byKey(String(p.key || ''));
    return k != null ? k : curOf(s);
  }));
  try { CACHE().put('TH_' + id, JSON.stringify(thRow_(row)), 21600); } catch (e) { /* 무시 */ }
  if (r && row.every((v, i) => v === (cur[i] || ''))) return;
  if (r) sh.getRange(r, 1, 1, TH_N).setNumberFormat('@').setValues([row]);
  else appendRow_('thumbs', row);
}

let LOCK_DEPTH = 0;

/* ---- 속도: 한 번의 실행 안에서는 같은 시트·폴더·속성을 다시 조회하지 않음 ---- */
const RT = { props: null, db: null, cap: null, root: null, folders: {} };
function P_(k) {
  if (!RT.props) RT.props = PropertiesService.getScriptProperties().getProperties();
  return RT.props[k] || null;
}
function setP_(k, v) {
  PropertiesService.getScriptProperties().setProperty(k, v);
  if (RT.props) RT.props[k] = v;
}
const CACHE = () => CacheService.getScriptCache();
function bustBoot_() { try { CACHE().remove('BOOT'); } catch (e) { /* 무시 */ } }
/** 스크립트 잠금 (중첩 호출은 바깥 잠금을 그대로 사용) */
function withLock_(fn) {
  if (LOCK_DEPTH > 0) return fn();
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(30000)) throw new Error('다른 사용자의 작업이 많습니다. 잠시 후 다시 시도하세요.');
  LOCK_DEPTH++;
  try {
    return fn();
  } finally {
    LOCK_DEPTH--;
    SpreadsheetApp.flush();
    lock.releaseLock();
  }
}

function stamp_(code) { return PropertiesService.getScriptProperties().getProperty('ST_' + code) || '0'; }
function touch_(code) {
  if (code) PropertiesService.getScriptProperties().setProperty('ST_' + code, String(Date.now()) + Math.random().toString(36).slice(2, 6));
}

/** 폴더 ID를 속성에 기억 → 다음부터는 ID로 바로 (없어졌으면 다시 찾기·만들기) */
function cachedFolder_(key, make) {
  if (RT.folders[key]) return RT.folders[key];
  const id = P_(key);
  if (id) {
    try { const f = DriveApp.getFolderById(id); if (!f.isTrashed()) return (RT.folders[key] = f); } catch (e) { /* 다시 */ }
  }
  const f = withLock_(make);
  setP_(key, f.getId());
  return (RT.folders[key] = f);
}

function trashFile_(fileId) {
  try { DriveApp.getFileById(fileId).moveTo(trashFolder_()); } catch (e) { /* 이미 없음 */ }
}

function getRootFolder_() {
  if (RT.root) return RT.root;
  if (CONFIG.ROOT_FOLDER_ID) return (RT.root = DriveApp.getFolderById(CONFIG.ROOT_FOLDER_ID));
  const id = P_('ROOT_FOLDER_ID');
  if (id) {
    try { return (RT.root = DriveApp.getFolderById(id)); } catch (e) { /* 새로 */ }
  }
  return (RT.root = withLock_(() => {
    const id2 = PropertiesService.getScriptProperties().getProperty('ROOT_FOLDER_ID');
    if (id2 && id2 !== id) return DriveApp.getFolderById(id2);
    const f = DriveApp.createFolder(CONFIG.ROOT_FOLDER_NAME);
    setP_('ROOT_FOLDER_ID', f.getId());
    return f;
  }));
}

/** 하위 폴더 찾기/만들기 — 동시에 두 명이 만들어도 하나만 생기도록 잠금 안에서 */
function getOrCreateSub_(parent, name) {
  return withLock_(() => {
    const it = parent.getFoldersByName(name);
    return it.hasNext() ? it.next() : parent.createFolder(name);
  });
}

function rowTo_(cols, r) {
  const o = {};
  cols.forEach((k, i) => {
    let v = r[i] == null ? '' : String(r[i]);
    if (k in JSONF) { try { v = v ? JSON.parse(v) : JSON.parse(JSON.stringify(JSONF[k])); } catch (e) { v = JSON.parse(JSON.stringify(JSONF[k])); } }
    o[k] = v;
  });
  o.ver = +o.ver || 0;
  return o;
}
function toRow_(cols, o) { return cols.map(k => (k in JSONF) ? JSON.stringify(o[k] == null ? JSONF[k] : o[k]) : (o[k] == null ? '' : String(o[k]))); }

/** 공종별 템플릿 (현장앱_DB '제원템플릿' 탭 — 사무실에서 시트로 직접 고쳐도 됨) */
function getTemplates_() {
  let rows = readTab_('templates');
  if (!rows.length) {
    withLock_(() => {
      if (readTab_('templates').length) return;
      const seed = [];
      SPEC_CATS.forEach(c => (SPEC_DEFAULTS[c] || []).forEach((x, i) => seed.push([c, i + 1, x[0], x[1], x[2], '일치', '', ''])));
      writeTab_('templates', seed);
    });
    rows = readTab_('templates');
  }
  if (P_('TPL_V') !== TPL_VER) rows = upgradeTemplates_(rows);
  const t = {};
  SPEC_CATS.forEach(c => { t[c] = []; });
  rows.sort((a, b) => (+a[1] || 0) - (+b[1] || 0)).forEach(r => {
    if (!r[0] || !r[2]) return;
    (t[r[0]] = t[r[0]] || []).push({ label: r[2], unit: r[3], type: r[4] === '문자' ? 'text' : 'num', mode: MODE_KEYS[r[5]] || 'eq', tol: r[6], note: r[7] });
  });
  return t;
}

/** 이 제원의 항목 구성을 공종 템플릿으로 저장 (기준값은 저장하지 않음) */
/** 양식 바꾸기 (v3): 손대지 않은 예전 기본 양식은 새 양식으로, 사무실에서 고친 양식은 고친 것을 두고 빠진 새 항목만 앞에 */
const TPL_VER = '3';
const TPL_OLD = {
  '흙막이 가시설': [['공법', '굴착 깊이', '엄지말뚝 규격', '엄지말뚝 간격', '띠장 규격', '버팀보 규격', '버팀보 수평 간격', '버팀보 단 간격 (수직)', '어스앵커 간격']],

  '합벽지지대': [['지지대 규격', '설치 간격', '설치 높이', '앵커 규격', '앵커 간격', '앵커 매입 깊이']]
};
const TPL_SAME = { '흙막이 공법': '공법', '최대 굴착 깊이': '굴착 깊이', 'SLAB 두께': '슬래브 두께', '높이': '설치 높이' };   // 이미 있으면 안 넣음
function upgradeTemplates_(rows) {
  return withLock_(() => {
    if (P_('TPL_V') === TPL_VER) return readTab_('templates');
    rows = readTab_('templates').filter(r => r[0] !== '거푸집 및 동바리');      // 시스템동바리로 합침
    Object.keys(TPL_OLD).forEach(cat => {
      const mine = rows.filter(r => r[0] === cat).sort((a, b) => (+a[1] || 0) - (+b[1] || 0)), labels = mine.map(r => String(r[2]));
      const def = SPEC_DEFAULTS[cat].map(x => [cat, 0, x[0], x[1], x[2], '일치', '', '']);
      let out;
      const isDefault = !mine.length || TPL_OLD[cat].some(o => o.join('|') === labels.join('|')) || def.map(r => r[2]).join('|') === labels.join('|');
      if (isDefault) out = def;
      else out = def.filter(r => labels.indexOf(r[2]) < 0 && !(TPL_SAME[r[2]] && labels.indexOf(TPL_SAME[r[2]]) >= 0)).concat(mine);
      rows = rows.filter(r => r[0] !== cat).concat(out.map((r, i) => { r = r.slice(); r[1] = i + 1; return r; }));
    });
    writeTab_('templates', rows);
    setP_('TPL_V', TPL_VER);
    bustBoot_();
    return readTab_('templates');
  });
}

function saveTemplate(cat, items, user) {
  user = needUser_(user);
  if (!cat) throw new Error('공종이 없습니다.');
  return withLock_(() => {
    const keep = readTab_('templates').filter(r => r[0] !== cat);
    const rows = (items || []).filter(x => x && String(x.label || '').trim()).map((x, i) =>
      [cat, i + 1, String(x.label).trim(), x.unit || '', x.type === 'text' ? '문자' : '수치', MODES[x.mode] || '일치', x.tol || '', x.note || '']);
    writeTab_('templates', keep.concat(rows));
    log_(user, '템플릿저장', { id: '', code: '', insp: '' }, `${cat} · ${rows.length}항목`, '');
    bustBoot_();
    return rows.length;
  });
}

function cleanItems_(items) {
  return (items || []).slice(0, 120).map(x => ({
    k: String(x.k || Utilities.getUuid().slice(0, 8)).slice(0, 40),
    label: String(x.label || '').trim().slice(0, 80), unit: String(x.unit || '').trim().slice(0, 20),
    type: x.type === 'text' ? 'text' : 'num', std: String(x.std == null ? '' : x.std).trim().slice(0, 80),
    mode: MODES[x.mode] ? x.mode : 'eq', tol: String(x.tol == null ? '' : x.tol).trim().slice(0, 20),
    note: String(x.note || '').trim().slice(0, 200)
  })).filter(x => x.label);
}

/** 변경 내용 요약 (이력용) */
function specDiff_(a, b) {
  const out = [];
  ['name', 'loc', 'doc', 'memo'].forEach(k => {
    if ((a[k] || '') !== (b[k] || '')) out.push(`${{ name: '대상명', loc: '설치 위치', doc: '기준 문서', memo: '비고' }[k]}: ${a[k] || '-'} → ${b[k] || '-'}`);
  });
  const am = {}, bm = {};
  (a.items || []).forEach(x => { am[x.k] = x; });
  (b.items || []).forEach(x => { bm[x.k] = x; });
  (b.items || []).forEach(x => {
    const o = am[x.k];
    if (!o) { out.push(`+ ${x.label} ${x.std}${x.unit}`); return; }
    const ch = [];
    if (o.label !== x.label) ch.push(`이름 ${o.label}→${x.label}`);
    if (o.std !== x.std) ch.push(`기준 ${o.std || '-'}→${x.std || '-'}${x.unit}`);
    if (o.unit !== x.unit) ch.push(`단위 ${o.unit || '-'}→${x.unit || '-'}`);
    if (o.mode !== x.mode) ch.push(`판정 ${MODES[o.mode]}→${MODES[x.mode]}`);
    if (o.tol !== x.tol) ch.push(`허용오차 ${o.tol || '없음'}→${x.tol || '없음'}`);
    if (ch.length) out.push(`${x.label}: ${ch.join(', ')}`);
  });
  (a.items || []).forEach(x => { if (!bm[x.k]) out.push(`- ${x.label}`); });
  return out.join(' / ');
}

/** 제원 저장 (새로 / 수정 — 수정은 버전 확인, 변경 내용은 작업이력에) */
function saveSpec(sp, user) {
  user = needUser_(user);
  if (sp.cat === '거푸집 및 동바리') sp.cat = '시스템동바리';
  if (!SPEC_CATS.includes(sp.cat)) throw new Error('공종이 올바르지 않습니다.');
  if (!String(sp.name || '').trim()) throw new Error('대상명을 입력하세요.');
  return withLock_(() => {
    const now = fmt_(new Date(), 'yyyy-MM-dd HH:mm:ss');
    if (!sp.id && sp.newId && /^[\w-]{8,64}$/.test(String(sp.newId))) {     // 오프라인 대기분 재전송
      const g0 = getRec_('specs', sp.newId);
      if (g0 && g0.o.by === user && !g0.o.del) { sp.id = sp.newId; sp.ver = g0.o.ver; }
    }
    const g = sp.id ? getRec_('specs', sp.id) : null;
    if (g && g.o.del) throw new Error('DELETED:다른 사용자가 이미 삭제한 제원입니다.');
    if (g && +sp.ver !== +g.o.ver) throw new Error(`CONFLICT:${g.o.mby}님이 먼저 수정했습니다 (${g.o.mat}).`);
    const o = g ? Object.assign({}, g.o) : { id: sp.newId && /^[\w-]{8,64}$/.test(String(sp.newId)) ? String(sp.newId) : Utilities.getUuid(), code: sp.code, cat: sp.cat, by: user, at: now, ver: 0, del: '' };
    const before = g ? JSON.parse(JSON.stringify(g.o)) : null;
    o.name = String(sp.name).trim().slice(0, 80);
    o.loc = String(sp.loc || '').trim().slice(0, 120);
    o.doc = String(sp.doc || '').trim().slice(0, 120);
    o.memo = String(sp.memo || '').trim().slice(0, 500);
    o.items = cleanItems_(sp.items);
    o.mby = user; o.mat = now; o.ver = (+o.ver || 0) + 1;
    putRec_('specs', g ? g.row : 0, o);
    log_(user, g ? '제원수정' : '제원등록', { id: o.id, code: o.code, insp: '' },
      g ? (`[v${o.ver}] ` + (specDiff_(before, o) || '변경 없음')) : `${o.cat} · ${o.name} · ${o.items.length}항목`, '');
    touch_(o.code);
    return o;
  });
}

/** 판정 (서버·화면 같은 규칙) */
function judge_(it, val) {
  const has = v => String(v == null ? '' : v).trim() !== '';
  const num = v => { const n = parseFloat(String(v).replace(/,/g, '')); return isFinite(n) ? n : null; };
  if (!has(val)) return { r: '미입력', why: '' };
  if (it.mode === 'rec') return { r: '확인 필요', why: '판정 기준 없는 기록 항목' };
  if (!has(it.std)) return { r: '확인 필요', why: '기준값 없음' };
  if (it.type === 'text') {
    const n = s => String(s).replace(/\s+/g, '').toUpperCase();
    return n(val) === n(it.std) ? { r: '일치', why: '' } : { r: '불일치', why: '기준과 다름' };
  }
  const s = num(it.std), v = num(val);
  if (s === null || v === null) return { r: '확인 필요', why: '숫자로 비교할 수 없음' };
  const t = has(it.tol) ? num(it.tol) : null;
  if (has(it.tol) && t === null) return { r: '확인 필요', why: '허용오차가 숫자가 아님' };
  const d = Math.round((v - s) * 1000) / 1000;
  if (it.mode === 'max') return v <= s + (t || 0) ? { r: '일치', why: d < 0 ? '기준 이하' : (d > 0 ? '허용오차 이내' : '') } : { r: '기준 초과', why: `기준 ${it.std}${t ? '+' + t : ''} 초과` };
  if (it.mode === 'min') return v >= s - (t || 0) ? { r: '일치', why: d > 0 ? '기준 이상' : (d < 0 ? '허용오차 이내' : '') } : { r: '기준 초과', why: `기준 ${it.std}${t ? '-' + t : ''} 미달` };
  if (d === 0) return { r: '일치', why: '' };
  if (t === null) return { r: '불일치', why: '허용오차 미설정 — 적합 여부는 확인 필요' };
  return Math.abs(d) <= t ? { r: '일치', why: `허용오차 ±${t} 이내` } : { r: '기준 초과', why: `허용오차 ±${t} 초과` };
}
function summarize_(items, vals) {
  const c = { total: items.length, '일치': 0, '불일치': 0, '기준 초과': 0, '미입력': 0, '확인 필요': 0 };
  items.forEach(it => { c[judge_(it, (vals[it.k] || {}).v).r]++; });
  return c;
}

/** 실측 저장 — 처음 저장할 때 제원의 기준을 함께 보관 */
function saveMeasure(m, user) {
  user = needUser_(user);
  return withLock_(() => {
    const now = fmt_(new Date(), 'yyyy-MM-dd HH:mm:ss');
    const g = m.id ? getRec_('measures', m.id) : null;
    if (g && m.isNew && g.o.by === user && !g.o.del) m.ver = g.o.ver;        // 오프라인 대기분 재전송
    if (g && g.o.del) throw new Error('DELETED:다른 사용자가 이미 삭제한 실측입니다.');
    if (g && +m.ver !== +g.o.ver) throw new Error(`CONFLICT:${g.o.mby}님이 먼저 수정했습니다 (${g.o.mat}).`);
    const sp = getRec_('specs', g ? g.o.specId : m.specId);
    if (!g && (!sp || sp.o.del)) throw new Error('제원을 찾을 수 없습니다.');
    const o = g ? Object.assign({}, g.o) : {
      id: /^[\w-]{8,64}$/.test(String(m.id || '')) ? String(m.id) : Utilities.getUuid(), code: sp.o.code, specId: sp.o.id,
      by: user, at: now, ver: 0, del: ''
    };
    if (!g || m.refresh) { o.items = JSON.parse(JSON.stringify(sp.o.items)); o.sver = sp.o.ver; }   // 기준 보관(또는 최신 제원 다시 적용)
    o.insp = String(m.insp || '').slice(0, 40);
    o.date = String(m.date || fmt_(new Date(), 'yyyy-MM-dd')).slice(0, 10);
    const vals = {};
    (o.items || []).forEach(it => {
      const x = (m.vals || {})[it.k] || {};
      vals[it.k] = { v: String(x.v == null ? '' : x.v).trim().slice(0, 40), note: String(x.note || '').trim().slice(0, 200), photos: (x.photos || []).filter(Boolean).slice(0, 20), same: x.same ? 1 : 0 };
    });
    o.vals = vals;
    o.sum = summarize_(o.items, vals);
    o.mby = user; o.mat = now; o.ver = (+o.ver || 0) + 1;
    putRec_('measures', g ? g.row : 0, o);
    log_(user, g ? '실측수정' : '실측등록', { id: o.id, code: o.code, insp: o.insp },
      `${sp ? sp.o.name : ''} · 일치 ${o.sum['일치']} 불일치 ${o.sum['불일치']} 기준초과 ${o.sum['기준 초과']} 미입력 ${o.sum['미입력']} 확인필요 ${o.sum['확인 필요']}`, '');
    touch_(o.code);
    return o;
  });
}

/** 제원·실측 삭제/복구/영구삭제 (실측에서 찍은 사진도 함께) */
function deleteRecs(tab, ids, vers, user, pin) {
  user = needUser_(user);
  if (!KIND[tab]) throw new Error('종류 오류');
  const admin = isAdmin_(pin);
  return withLock_(() => {
    const out = [], codes = {};
    const now = fmt_(new Date(), 'yyyy-MM-dd HH:mm:ss');
    (ids || []).forEach(id => {
      const g = getRec_(tab, id);
      if (!g) return out.push({ id, r: 'none' });
      const o = g.o;
      if (o.del) return out.push({ id, r: 'already', by: o.dby });
      if (!admin && o.by !== user) return out.push({ id, r: 'denied', by: o.by });
      if (vers && vers[id] != null && +vers[id] !== +o.ver) return out.push({ id, r: 'changed', by: o.mby });
      if (tab === 'specs' && readRecs_('measures').some(m => m.specId === id && !m.del)) return out.push({ id, r: 'used' });
      o.del = 'Y'; o.dby = user; o.dat = now; o.ver = (+o.ver || 0) + 1;
      putRec_(tab, g.row, o);
      if (tab === 'measures') {
        readTab_('items').map(rowToItem_).filter(x => x.sid === id && x.kind === '실측' && !x.del).forEach(x => {
          const gi = getItem_(x.id);
          [x.bId, x.aId].forEach(f => { if (f && !isPending_(f)) trashFile_(f); });
          x.del = 'Y'; x.dby = user; x.dat = now; x.ver = (+x.ver || 0) + 1;
          putItem_(gi.row, x);
        });
      }
      log_(user + (admin ? ' (관리자)' : ''), tab === 'specs' ? '제원삭제' : '실측삭제', { id, code: o.code, insp: o.insp || '' }, o.name || '', '');
      codes[o.code] = 1;
      out.push({ id, r: 'ok' });
    });
    Object.keys(codes).forEach(touch_);
    return out;
  });
}
function restoreRecs(tab, ids, user, pin) {
  user = needUser_(user);
  if (!KIND[tab]) throw new Error('종류 오류');
  const admin = isAdmin_(pin);
  return withLock_(() => {
    const out = [], codes = {};
    (ids || []).forEach(id => {
      const g = getRec_(tab, id);
      if (!g) return out.push({ id, r: 'none' });
      const o = g.o;
      if (!o.del) return out.push({ id, r: 'notdeleted' });
      if (o.del === '영구') return out.push({ id, r: 'purged' });
      if (!admin && o.by !== user && o.dby !== user) return out.push({ id, r: 'denied' });
      const when = o.dat;
      o.del = ''; o.dby = ''; o.dat = ''; o.ver = (+o.ver || 0) + 1; o.mby = user; o.mat = fmt_(new Date(), 'yyyy-MM-dd HH:mm:ss');
      putRec_(tab, g.row, o);
      if (tab === 'measures') {
        const folder = itemPhotoFolder_(o.code, o.insp || '제원실측');
        readTab_('items').map(rowToItem_).filter(x => x.sid === id && x.kind === '실측' && x.del === 'Y' && x.dat === when).forEach(x => {
          const gi = getItem_(x.id);
          [x.bId].forEach(f => { if (f && !isPending_(f)) { try { DriveApp.getFileById(f).moveTo(folder); } catch (e) { /* 없음 */ } } });
          x.del = ''; x.dby = ''; x.dat = ''; x.ver = (+x.ver || 0) + 1;
          putItem_(gi.row, x);
        });
      }
      log_(user, tab === 'specs' ? '제원복구' : '실측복구', { id, code: o.code, insp: o.insp || '' }, o.name || '', '');
      codes[o.code] = 1;
      out.push({ id, r: 'ok' });
    });
    Object.keys(codes).forEach(touch_);
    return out;
  });
}
function purgeRecs(tab, ids, user, pin) {
  user = needUser_(user);
  if (!isAdmin_(pin)) throw new Error('관리자만 영구 삭제할 수 있습니다.');
  return withLock_(() => (ids || []).map(id => {
    const g = getRec_(tab, id);
    if (!g || g.o.del !== 'Y') return { id, r: 'skip' };
    g.o.del = '영구'; g.o.ver = (+g.o.ver || 0) + 1;
    putRec_(tab, g.row, g.o);
    if (tab === 'measures') {
      const its = readTab_('items').map(rowToItem_).filter(x => x.sid === id && x.kind === '실측' && x.del === 'Y').map(x => x.id);
      if (its.length) purgeItems(its, user, pin);
    }
    log_(user + ' (관리자)', tab === 'specs' ? '제원영구삭제' : '실측영구삭제', { id, code: g.o.code, insp: '' }, '', '');
    return { id, r: 'ok' };
  }));
}

/** 비교표 출력 (엑셀 + PDF) — 실측 여러 건 + 불일치 모음 */
function exportCompare(code, ids, user) {
  user = needUser_(user);

  const ms = readRecs_('measures').filter(m => m.code === code && !m.del && (!ids || !ids.length || ids.includes(m.id)))
    .sort((a, b) => (a.date + a.at < b.date + b.at ? -1 : 1));

}

/* ======================= 자동 백업 ======================= */
// 처음 한 번: 편집기에서 setupBackup 실행 → 매일 새벽 2시 자동 백업 (최근 BACKUP_KEEP 개 보관)
const BACKUP_KEEP = 30;
const BACKUP_PREFIX = '현장앱_DB_백업_';

function backupFolder_() { return cachedFolder_('FD_BACKUP', () => getOrCreateSub_(getRootFolder_(), '_백업')); }

/** 트리거 설치 (편집기에서 한 번 실행) */
function setupBackup() {
  ScriptApp.getProjectTriggers().filter(t => t.getHandlerFunction() === 'backupDb').forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('backupDb').timeBased().atHour(2).everyDays(1).inTimezone(TZ).create();
  const r = doBackup_('처음 설정');
  Logger.log('자동 백업 설정 완료 — 매일 새벽 2시. 첫 백업: ' + r.name);
  return r;
}

/** 매일 트리거가 부르는 함수 */
function backupDb() { return doBackup_('자동'); }

function doBackup_(why) {
  const folder = backupFolder_();
  const stamp = fmt_(new Date(), 'yyyy-MM-dd_HHmm');
  const db = DriveApp.getFileById(getDb_().getId()).makeCopy(BACKUP_PREFIX + stamp, folder);
  db.setDescription('백업:' + why);
  try { DriveApp.getFileById(getCaptionSS_().getId()).makeCopy('현장앱_기본캡션_백업_' + stamp, folder); } catch (e) { /* 캡션 없음 */ }
  // 오래된 것 정리 (DB·캡션 각각 최근 BACKUP_KEEP 개)
  [BACKUP_PREFIX, '현장앱_기본캡션_백업_'].forEach(pre => {
    const fs = [], it = folder.getFiles();
    while (it.hasNext()) { const f = it.next(); if (f.getName().indexOf(pre) === 0) fs.push(f); }
    fs.sort((a, b) => b.getName() < a.getName() ? -1 : 1);       // 이름이 날짜순
    fs.slice(BACKUP_KEEP).forEach(f => { try { f.setTrashed(true); } catch (e) { /* 무시 */ } });
  });
  const at = fmt_(new Date(), 'yyyy-MM-dd HH:mm');
  setP_('LAST_BACKUP', at + ' (' + why + ')');
  return { id: db.getId(), name: db.getName(), at };
}

function needAdmin_(pin) { if (!isAdmin_(pin)) throw new Error('관리자 PIN이 맞지 않습니다.'); }

/** 관리자: 지금 백업 */
function backupNow(user, pin) {
  needAdmin_(pin); user = needUser_(user);
  const r = doBackup_('수동 · ' + user);
  log_(user, '백업', {}, r.name, r.id);
  return r;
}

/** 관리자: 백업 목록 */
function listBackups(user, pin) {
  needAdmin_(pin);
  const folder = backupFolder_(), out = [], it = folder.getFiles();
  while (it.hasNext()) {
    const f = it.next();
    if (f.getName().indexOf(BACKUP_PREFIX) !== 0) continue;
    out.push({ id: f.getId(), name: f.getName(), why: String(f.getDescription() || '').replace(/^백업:/, ''),
      at: fmt_(f.getDateCreated(), 'yyyy-MM-dd HH:mm'), url: f.getUrl() });
  }
  out.sort((a, b) => b.name < a.name ? -1 : 1);
  let auto = false;
  try { auto = ScriptApp.getProjectTriggers().some(t => t.getHandlerFunction() === 'backupDb'); } catch (e) { /* 권한 없음 */ }
  return { list: out, auto, last: P_('LAST_BACKUP') || '', keep: BACKUP_KEEP, folder: folder.getUrl() };
}

/** 관리자: 백업으로 되돌리기 — 지금 상태를 먼저 백업한 뒤, 백업 사본을 새 DB로 사용 */
function restoreBackup(fileId, user, pin) {
  needAdmin_(pin); user = needUser_(user);
  return withLock_(() => {
    const src = DriveApp.getFileById(fileId);
    if (src.getName().indexOf(BACKUP_PREFIX) !== 0) throw new Error('백업 파일이 아닙니다.');
    const before = doBackup_('복원 직전 · ' + user);
    const curDb = getDb_();
    // 파일번호는 지금 값과 백업 값 중 큰 값 유지 (지운 번호·새 사진 번호를 다시 쓰지 않도록)
    const cnt = {};
    try { readTab_('counters').forEach(r => { cnt[r[0]] = +r[1] || 0; }); } catch (e) { /* 없음 */ }
    const copy = src.makeCopy('현장앱_DB', getRootFolder_());
    copy.setDescription('복원: ' + src.getName() + ' · ' + user);
    const old = DriveApp.getFileById(curDb.getId());
    old.setName('현장앱_DB_복원전_' + fmt_(new Date(), 'yyyy-MM-dd_HHmm'));
    old.moveTo(backupFolder_());
    setP_('DB_ID', copy.getId());
    RT.db = null; Object.keys(TAB_CACHE).forEach(k => delete TAB_CACHE[k]);
    const rows = readTab_('counters');
    rows.forEach(r => { if (cnt[r[0]] > (+r[1] || 0)) r[1] = String(cnt[r[0]]); delete cnt[r[0]]; });
    Object.keys(cnt).forEach(k => rows.push([k, String(cnt[k])]));
    writeTab_('counters', rows);
    // 모든 휴대폰이 목록을 새로 받도록
    const codes = new Set(readTab_('svc').map(r => r[0]).concat(readTab_('items').map(r => r[1])));
    codes.forEach(c => touch_(c));
    bustBoot_();
    log_(user, '복원', {}, src.getName() + ' (복원 직전 백업: ' + before.name + ')', copy.getId());
    return { ok: true, from: src.getName(), before: before.name };
  });
}


/* ======================= 통합관리 엑셀 자동 반영 =======================
 * 구글 드라이브(PC의 G: › 내 드라이브)에 있는 「안전점검 통합관리」 엑셀을 30분마다 확인해서
 * 저장된 내용이 바뀌었으면 「현장앱 보내기」와 똑같이 반영. (엑셀 매크로가 읽는 칸과 같은 칸을 읽음)
 */
const SYNC_NAME = '통합관리';                 // 파일 이름에 이 글자가 들어 있는 엑셀(.xlsm/.xlsx)을 찾음
const SYNC_EVERY_MIN = 30;

function syncFile_() {
  const id = P_('SYNC_ID');
  if (id) { try { const f = DriveApp.getFileById(id); if (!f.isTrashed()) return f; } catch (e) { /* 지워짐 → 다시 찾기 */ } }
  // 엑셀 형식만 찾아서 빠르게 (드라이브 전체를 훑지 않도록 최대 50개까지만)
  const it = DriveApp.searchFiles(`title contains '${SYNC_NAME}' and trashed = false and (mimeType = 'application/vnd.ms-excel.sheet.macroEnabled.12'`
    + ` or mimeType = 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet' or mimeType = 'application/vnd.ms-excel')`);
  let best = null, n = 0;
  while (it.hasNext() && n++ < 50) {
    const f = it.next();
    if (!/\.xls[xm]$/i.test(f.getName())) continue;
    if (!best || f.getLastUpdated() > best.getLastUpdated()) best = f;
  }
  if (best) setP_('SYNC_ID', best.getId());
  return best;
}
/** 상태 화면이 드라이브를 다시 뒤지지 않도록 파일 정보를 기억 */
function rememberFile_(f) {
  if (!f) return;
  setP_('SYNC_FILE', JSON.stringify({ name: f.getName(), url: f.getUrl(), at: fmt_(f.getLastUpdated(), 'yyyy-MM-dd HH:mm') }));
}
/** 엑셀 → 구글 시트 임시 사본 (드라이브 변환) */
function convertXls_(file) {
  const tmp = cachedFolder_('FD_JOBS', () => getOrCreateSub_(getRootFolder_(), '_작업중'));
  const res = UrlFetchApp.fetch(`https://www.googleapis.com/drive/v3/files/${file.getId()}/copy?supportsAllDrives=true&fields=id`, {
    method: 'post', contentType: 'application/json', muteHttpExceptions: true,
    headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() },
    payload: JSON.stringify({ name: '_통합관리_읽기용_' + fmt_(new Date(), 'HHmmss'), mimeType: 'application/vnd.google-apps.spreadsheet', parents: [tmp.getId()] })
  });
  const j = JSON.parse(res.getContentText() || '{}');
  if (!j.id) throw new Error('엑셀을 읽지 못했습니다 (' + ((j.error && j.error.message) || res.getResponseCode()) + ')');
  return j.id;
}
const cleanX_ = v => {
  if (v == null || v === '') return '';
  const t = String(v);
  if (/^#(N\/A|NAME\?|REF!|VALUE!|DIV\/0!|NUM!|NULL!)/.test(t)) return '';
  return t.replace(/[\t\r\n]+/g, ' ').trim();
};
const dateX_ = v => {
  if (Object.prototype.toString.call(v) === '[object Date]' && !isNaN(v.getTime())) return fmt_(v, 'yyyy-MM-dd');
  if (typeof v === 'number' && v > 30000 && v < 80000) return fmt_(new Date(Math.round((v - 25569) * 86400000)), 'yyyy-MM-dd');
  return cleanX_(v);
};
/** 변환된 시트에서 「현장앱 보내기」 매크로와 같은 칸 읽기 */
function readTonghap_(ss) {
  const sh = n => { const x = ss.getSheetByName(n); if (!x) throw new Error(`「${n}」 시트가 없습니다`); return x; };
  const svc = [], items = [], chk = [], codes = {};
  sh('용역관리').getRange('A7:AC106').getValues().forEach(v => {
    const code = cleanX_(v[28]);
    if (code && cleanX_(v[1])) { codes[code] = 1; svc.push([code, cleanX_(v[1]), cleanX_(v[3]), cleanX_(v[5]), cleanX_(v[4]), cleanX_(v[9])]); }
  });
  const map = {};
  sh('기준_점검시기').getRange('G2:I204').getValues().forEach(k => { const a = cleanX_(k[0]); if (a) map[a] = [cleanX_(k[1]), cleanX_(k[2])]; });
  sh('점검이력').getRange('C7:T606').getValues().forEach(v => {
    const code = cleanX_(v[17]), item = cleanX_(v[1]);
    if (code && item && codes[code]) {
      const m = map[item] || ['', ''];
      items.push([code, cleanX_(v[0]), item, m[0], m[1], cleanX_(v[2]), dateX_(v[3]), cleanX_(v[5]), dateX_(v[6]), cleanX_(v[16])]);
    }
  });
  const ck = ss.getSheetByName('기준_공종별체크리스트');
  if (ck && ck.getLastRow() >= 2) ck.getRange(2, 2, ck.getLastRow() - 1, 3).getValues().forEach(v => {
    const f = cleanX_(v[0]);
    if (f && f[0] !== '※' && cleanX_(v[2])) chk.push([f, cleanX_(v[1]), cleanX_(v[2])]);
  });
  return { svc, items, chk };
}
/** 확인 후 반영. force: 바뀌지 않았어도 반영 */
function runSync_(force, how) {
  const f = syncFile_();
  if (!f) { setP_('SYNC_LAST', fmt_(new Date(), 'yyyy-MM-dd HH:mm') + ' · 통합관리 엑셀을 찾지 못함'); return { ok: false, msg: `드라이브에서 이름에 「${SYNC_NAME}」이 들어간 엑셀(.xlsm)을 찾지 못했습니다.` }; }
  rememberFile_(f);
  const mt = String(f.getLastUpdated().getTime());
  if (!force && P_('SYNC_MT') === mt) return { ok: true, same: true, msg: '바뀐 내용 없음', file: f.getName() };
  let tmpId = '';
  try {
    tmpId = convertXls_(f);
    const d = readTonghap_(SpreadsheetApp.openById(tmpId));
    // 안전장치: 읽은 용역·점검이 하나도 없으면 (칸이 바뀌었거나 수식 오류) 덮어쓰지 않음
    if (!d.svc.length || !d.items.length) throw new Error(`읽은 용역 ${d.svc.length}건 · 점검 ${d.items.length}건 — 시트 칸 위치를 확인하세요. 기존 내용은 그대로 둡니다.`);
    const msg = applyPush_(d.svc, d.items, d.chk, how || '자동 반영');
    setP_('SYNC_MT', mt);
    setP_('SYNC_LAST', fmt_(new Date(), 'yyyy-MM-dd HH:mm') + ' · ' + msg);
    return { ok: true, msg, file: f.getName() };
  } catch (e) {
    setP_('SYNC_LAST', fmt_(new Date(), 'yyyy-MM-dd HH:mm') + ' · 실패: ' + e.message);
    return { ok: false, msg: e.message, file: f.getName() };
  } finally {
    if (tmpId) { try { DriveApp.getFileById(tmpId).setTrashed(true); } catch (e) { /* 무시 */ } }
  }
}
/** 30분마다 트리거가 부름 */
function autoSync() { return runSync_(false, '자동 반영'); }
/** 편집기에서 한 번 실행해도 되고, 앱 관리자 화면에서 켜도 됨 */
function setupAutoSync() { setAutoSync_(true); return runSync_(true, '처음 반영'); }
function setAutoSync_(on) {
  ScriptApp.getProjectTriggers().filter(t => t.getHandlerFunction() === 'autoSync').forEach(t => ScriptApp.deleteTrigger(t));
  if (on) ScriptApp.newTrigger('autoSync').timeBased().everyMinutes(SYNC_EVERY_MIN).create();
  setP_('SYNC_AUTO', on ? 'Y' : 'N');
}

/* --- 관리자 화면용 --- */
/** 상태: 저장해 둔 정보만 읽음 (빠름). find=true 일 때만 드라이브에서 파일을 찾아봄 */
function syncStatus(user, pin, find) {
  needAdmin_(pin);
  let err = '';
  if (find || !P_('SYNC_FILE')) { try { rememberFile_(syncFile_()); } catch (e) { err = e.message; } }
  let file = null; try { file = JSON.parse(P_('SYNC_FILE') || 'null'); } catch (e) { file = null; }
  let auto = P_('SYNC_AUTO');
  if (!auto) { try { auto = ScriptApp.getProjectTriggers().some(t => t.getHandlerFunction() === 'autoSync') ? 'Y' : 'N'; setP_('SYNC_AUTO', auto); } catch (e) { auto = 'N'; } }
  return { auto: auto === 'Y', every: SYNC_EVERY_MIN, file, last: P_('SYNC_LAST') || '', lastPush: P_('LAST_PUSH') || '', err };
}
function syncNow(user, pin) { needAdmin_(pin); user = needUser_(user); const r = runSync_(true, '지금 반영 · ' + user); log_(user, '통합관리반영', {}, r.msg, ''); return r; }
function syncAuto(on, user, pin) { needAdmin_(pin); needUser_(user); setAutoSync_(!!on); return syncStatus(user, pin); }
/** 다른 파일로 연결 (드라이브 링크 붙여넣기) */
function syncSetFile(link, user, pin) {
  needAdmin_(pin); needUser_(user);
  const m = String(link || '').match(/[-\w]{25,}/);
  if (!m) throw new Error('드라이브 파일 링크가 아닙니다.');
  const f = DriveApp.getFileById(m[0]);
  if (!/\.xls[xm]$/i.test(f.getName())) throw new Error('엑셀 파일(.xlsm / .xlsx)을 골라 주세요.');
  setP_('SYNC_ID', f.getId()); setP_('SYNC_MT', ''); rememberFile_(f);
  return syncStatus(user, pin);
}
